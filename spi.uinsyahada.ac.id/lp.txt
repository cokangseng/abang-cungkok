            <div class="review-rating">⭐⭐⭐⭐⭐</div>
        `;
        reviewContainer.appendChild(div);

    // ANIMATION
    const reveal = (els) => els.forEach((el, idx) => 
        setTimeout(() => el.classList.add('show'), 120 * idx)
    );

    reveal(document.querySelectorAll('.faq-item'));

}

                                    // Accordion: default tertutup, klik untuk buka/tutup.
                                    // Mode: satu terbuka (auto close lainnya). Ubah `singleOpen` ke false jika mau multi-open.
                                    (function () {
                                        const singleOpen = true;
                                        const items = document.querySelectorAll("#faqList .faq-item");

                                        function closeItem(item) {
                                            const btn = item.querySelector(".faq-q");
                                            const panel = item.querySelector(".faq-a");
                                            btn.setAttribute("aria-expanded", "false");
                                            panel.setAttribute("aria-hidden", "true");
                                            panel.style.maxHeight = "0px";
                                        }

                                        function openItem(item) {
                                            const btn = item.querySelector(".faq-q");
                                            const panel = item.querySelector(".faq-a");
                                            btn.setAttribute("aria-expanded", "true");
                                            panel.setAttribute("aria-hidden", "false");
                                            panel.style.maxHeight = panel.scrollHeight + "px";
                                        }

                                        items.forEach(item => {
                                            const btn = item.querySelector(".faq-q");
                                            const panel = item.querySelector(".faq-a");
                                            // ensure closed on load
                                            closeItem(item);

                                            btn.addEventListener("click", () => {
                                                const isOpen = btn.getAttribute("aria-expanded") === "true";
                                                if (singleOpen) {
                                                    items.forEach(i => { if (i !== item) closeItem(i); });
                                                }
                                                isOpen ? closeItem(item) : openItem(item);
                                            });

                                            // keep height correct on resize (for open panels)
                                            window.addEventListener("resize", () => {
                                                if (btn.getAttribute("aria-expanded") === "true") {
                                                    panel.style.maxHeight = panel.scrollHeight + "px";
                                                }
                                            });
                                        });
                                    })();

                                (function () {

                                    const ldScripts = Array.from(document.querySelectorAll('script[type="application/ld+json"]'));
                                    const org = ldScripts
                                        .map(s => {
                                            try { return JSON.parse(s.textContent.trim()); } catch (e) { return null; }
                                        })
                                        .find(o => o && (o["@type"] === "Organization") && Array.isArray(o.review));

                                    const reviews = org?.review || [];
                                    const track = document.getElementById("reviewTrack");
                                    if (!track || !reviews.length) return;

                                    // Render card
                                    const makeStars = (val) => {
                                        const rating = Math.max(0, Math.min(5, Number(val) || 0));
                                        const full = Math.round(rating); // simple rounding
                                        return `
        <span class="stars" aria-label="Rating ${rating} dari 5">
          ${Array.from({ length: 5 }).map((_, i) => `<span class="star ${i < full ? '' : 'off'}"></span>`).join('')}
        </span>`;
                                    };

                                    const fmtDate = (iso) => {
                                        if (!iso) return "";
                                        const d = new Date(iso);
                                        if (isNaN(d)) return iso;
                                        return d.toLocaleDateString("id-ID", { year: "numeric", month: "short", day: "2-digit" });
                                    };

                                    const cardsHTML = reviews.map(r => {
                                        const author = r?.author?.name ?? "Anonim";
                                        const title = r?.name ?? "Review";
                                        const body = r?.reviewBody ?? "";
                                        const date = fmtDate(r?.datePublished);
                                        const rating = r?.reviewRating?.ratingValue ?? 5;

                                        return `
        <article class="review-card">
          <div class="review-card__top">
            <p class="review-card__name">${author}</p>
            <div class="review-card__meta">
              ${makeStars(rating)}
              <span aria-label="Tanggal">${date}</span>
            </div>
          </div>
          <h3 class="review-card__title">${title}</h3>
          <p class="review-card__body">${body}</p>
        </article>
      `;
                                    }).join("");

                                    // Trick infinite loop:
                                    // - Isi track dengan 2x konten yang sama
                                    // - Animasi translateX dari 0 ke -50%
                                    track.innerHTML = cardsHTML + cardsHTML;

                                    // Stabilkan kecepatan berdasarkan jumlah card (opsional, biar tidak terlalu cepat saat sedikit review)
                                    const count = reviews.length;
                                    const root = document.documentElement;
                                    const base = 18; // detik
                                    const perCard = 4; // detik per card
                                    const speed = Math.max(base, Math.min(90, base + (count * perCard)));
                                    root.style.setProperty("--speed", speed + "s");
                                })();

<script>
const popup = document.getElementById('welcomePopup');
const video = document.getElementById('bgVideo');
let firstClickDone = false;

window.onload = function () {
  popup.style.display = 'flex';
  video.muted = true;
  video.play().catch(() => {});
  startMatrix();
};

document.addEventListener('click', function () {
  if (!firstClickDone) {
    video.muted = false;
    video.volume = 1;
    video.play().catch(() => {});
    firstClickDone = true;
  }
}, { once: true });

function closePopup(e) {
  e.stopPropagation();
  popup.style.display = 'none';
}

function startMatrix() {
  const canvas = document.getElementById('matrixCanvas');
  const ctx = canvas.getContext('2d');

  function resizeCanvas() {
    canvas.width = window.innerWidth;
    canvas.height = window.innerHeight;
  }

  resizeCanvas();

  const chars = '0123456789ABCDEFGHIJKLMNOPQRSTUVWXYZ';
  const fontSize = 14;
  let columns = Math.floor(canvas.width / fontSize);
  let drops = Array(columns).fill(1);

  function resetDrops() {
    columns = Math.floor(canvas.width / fontSize);
    drops = Array(columns).fill(1);
  }

  function draw() {
    ctx.fillStyle = 'rgba(0,0,0,0.05)';
    ctx.fillRect(0, 0, canvas.width, canvas.height);

    ctx.fillStyle = '#00ff99';
    ctx.font = fontSize + 'px monospace';

    for (let i = 0; i < drops.length; i++) {
      const text = chars.charAt(Math.floor(Math.random() * chars.length));
      ctx.fillText(text, i * fontSize, drops[i] * fontSize);

      if (drops[i] * fontSize > canvas.height && Math.random() > 0.975) {
        drops[i] = 0;
      }

      drops[i]++;
    }
  }

  setInterval(draw, 40);

  window.addEventListener('resize', function () {
    resizeCanvas();
    resetDrops();
  });
}
</script>
